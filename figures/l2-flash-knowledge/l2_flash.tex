% L2 (Gwinnett R3) recovery vs. knowledge — vector twin of l2_flash.png.
% Reads l2_flash.dat (written by l2_flash.py). Preamble is scaffold/main.tex.
% Bars: rows 1-3 (K1,K2,K3) clear the join-only null -> c0; row 0 -> gray.
% Null corrected 2026-09-23 to 3.35% (results/join_null_2026-09-23.txt; was 5.87%).
\begin{tikzpicture}
\begin{axis}[
  width=8.8cm, height=6.0cm,
  ybar, bar width=20pt,
  axis lines*=left,
  xmin=-0.6, xmax=3.6, ymin=0, ymax=90,
  xtick={0,1,2,3}, xticklabels={K0, K1, K2, K3},
  ytick={0,25,50,75}, yticklabel={\pgfmathprintnumber{\tick}\%},
  xlabel={Adversary Knowledge Level},
  ylabel={Ballot Recovered (\% of Scored Attempts)},
  ymajorgrids, grid style={black!12, line width=0.4pt},
  tick style={black!55}, tick align=outside,
  legend style={at={(0.03,0.97)}, anchor=north west, draw=none, fill=none,
                cells={anchor=west}, font=\small},
  clip=false,
]
  % reference line (\draw, not \addplot: the axis is in ybar mode, so an
  % \addplot line would be rendered as bars at the plot edges).
  % Only the join-only null is drawn: it IS the exploitation criterion here,
  % and a second line at 50% made the coincidental K2 = 50.0% read as the
  % threshold.
  \draw[black!55, dotted, line width=1.3pt]
    (axis cs:-0.6,3.35) -- (axis cs:3.6,3.35);
  \node[anchor=south west, font=\small, text=black!50]
    at (axis cs:-0.55,4.5) {null $=3.4\%$};
  % not distinguishable (K0,K1) — rows 0,1
  \addplot[ybar, bar width=20pt, bar shift=0pt, fill=black!30, draw=black!55,
           error bars/.cd, y dir=both, y explicit,
           error bar style={black!55, line width=0.8pt},
           error mark options={rotate=90, black!55, mark size=3pt}]
    table[x=kidx, y=rate, skip coords between index={1}{4},
          y error plus expr=\thisrow{hi}-\thisrow{rate},
          y error minus expr=\thisrow{rate}-\thisrow{lo}] {l2_flash.dat};
  % clears the null (K2,K3) — rows 2,3
  \addplot[ybar, bar width=20pt, bar shift=0pt, fill=c0, draw=black!55,
           error bars/.cd, y dir=both, y explicit,
           error bar style={black!55, line width=0.8pt},
           error mark options={rotate=90, black!55, mark size=3pt}]
    table[x=kidx, y=rate, skip coords between index={0}{1},
          y error plus expr=\thisrow{hi}-\thisrow{rate},
          y error minus expr=\thisrow{rate}-\thisrow{lo}] {l2_flash.dat};
  \legend{Not Distinguishable From Null, Clears Null (Exploited)}
\end{axis}
\end{tikzpicture}
